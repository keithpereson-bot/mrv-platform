updated++; break; } } }); bumpVersion_(); return { ok: true, updated: updated }; } // ── Mapping ──────────────────────────────────────────────── function mapRequestRow_(r) { return [ r.ref || '', r.project || '', r.type || '', r.category || '', r.costSummary || '', r.costItem || '', r.description || '', r.amount || 0, r.requester || '', r.status || '', r.date || '', r.sigLabel || '', r.sigTime || '', r.approvedBy || '', r.approvedAt || '', r.requesterSigName || '', r.requesterSigTime || '', r.clientSN || '', r.clientName || '', r.isExcess === true || r.isExcess === 'Yes' ? JSON.stringify(r.transportBreakdown) : (r.transportBreakdown || ''), r.payeeNumber || '', (r.sigDataUrl && String(r.sigDataUrl).length > 50) ? r.sigDataUrl : '', (r.requesterSig && String(r.requesterSig).length > 50) ? r.requesterSig : '' ]; } // ── Sheet helpers ────────────────────────────────────────── function ensureSheet_(ss, name, headers) { var sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.getRange(1, 1, 1, headers.length).setValues([headers]); sh.setFrozenRows(1); } return sh; } function migrateRequestHeaders_(ss) { var sh = ss.getSheetByName(SHEET.REQUESTS); if (!sh) { ensureSheet_(ss, SHEET.REQUESTS, REQ_HEADERS); return; } var lastCol = Math.max(sh.getLastColumn(), 1); if (!list || !list.length) return; var rows = list.map(mapper); if (!rows.length) return; sh.getRange(2, 1, rows.length, headers.length).setValues(rows); } // ── Version (for auto-sync) ──────────────────────────────── function getVersion_() { var cache = CacheService.getScriptCache();
